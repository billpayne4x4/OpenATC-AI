cmake_minimum_required(VERSION 3.24)
project(OpenATC VERSION 0.2.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
if(WIN32)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
option(OPENATC_BUILD_DESKTOP "Build standalone ImGui app" ON)
option(OPENATC_BUILD_ENGINE "Build HTTP engine" ON)
option(OPENATC_BUILD_PLUGIN "Build X-Plane compatibility plugin" OFF)
set(XPLANE_SDK "" CACHE PATH "Directory containing SDK/CHeaders and SDK/Libraries")
include(CTest)
add_library(openatc_core src/core.cpp src/flight.cpp src/airport_data.cpp)
target_include_directories(openatc_core PUBLIC include)
if(BUILD_TESTING)
  add_executable(openatc_tests tests/core_tests.cpp)
  target_link_libraries(openatc_tests PRIVATE openatc_core)
  add_test(NAME core COMMAND openatc_tests)
endif()
if(OPENATC_BUILD_ENGINE OR OPENATC_BUILD_DESKTOP OR OPENATC_BUILD_PLUGIN)
  include(FetchContent)
  find_package(Threads REQUIRED)
  find_package(OpenSSL REQUIRED)
  FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(json)
  set(HTTPLIB_REQUIRE_OPENSSL ON CACHE BOOL "" FORCE)
  set(HTTPLIB_USE_ZLIB_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
  set(HTTPLIB_USE_BROTLI_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(httplib URL https://github.com/yhirose/cpp-httplib/archive/refs/tags/v0.18.3.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(httplib)
  add_library(openatc_transport INTERFACE)
  target_link_libraries(openatc_transport INTERFACE openatc_core nlohmann_json::nlohmann_json httplib::httplib Threads::Threads)
  add_library(openatc_transport_plain INTERFACE)
  target_include_directories(openatc_transport_plain INTERFACE ${httplib_SOURCE_DIR})
  target_link_libraries(openatc_transport_plain INTERFACE openatc_core nlohmann_json::nlohmann_json Threads::Threads)
  if(WIN32)
    target_link_libraries(openatc_transport_plain INTERFACE ws2_32)
  endif()
endif()
if(OPENATC_BUILD_ENGINE)
  add_executable(open-atc-engine src/engine.cpp src/simbrief.cpp)
  target_link_libraries(open-atc-engine PRIVATE openatc_transport)
  install(TARGETS open-atc-engine RUNTIME DESTINATION bin)
  if(BUILD_TESTING)
    add_executable(openatc_integration_tests tests/integration_tests.cpp src/simbrief.cpp)
    target_link_libraries(openatc_integration_tests PRIVATE openatc_transport_plain)
    add_test(NAME parsing COMMAND openatc_integration_tests)
  endif()
endif()
if(OPENATC_BUILD_DESKTOP OR OPENATC_BUILD_PLUGIN)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    include(cmake/LinuxGraphics.cmake)
  else()
    find_package(OpenGL REQUIRED)
    set(OPENATC_GRAPHICS_TARGET OpenGL::GL)
  endif()
  FetchContent_Declare(imgui URL https://github.com/ocornut/imgui/archive/refs/tags/v1.91.9b.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE SOURCE_SUBDIR openatc-no-cmake)
  FetchContent_MakeAvailable(imgui)
  FetchContent_Declare(miniaudio URL https://github.com/mackron/miniaudio/archive/refs/tags/0.11.22.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE SOURCE_SUBDIR openatc-no-cmake)
  FetchContent_MakeAvailable(miniaudio)
  add_library(openatc_imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp)
  target_include_directories(openatc_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  set(OPENATC_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
  add_custom_command(OUTPUT "${OPENATC_GENERATED_DIR}/openatc_fonts.hpp"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed_fonts.py" "${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts" "${OPENATC_GENERATED_DIR}/openatc_fonts.hpp"
    DEPENDS scripts/embed_fonts.py assets/fonts/DejaVuSans.ttf assets/fonts/DejaVuSans-Bold.ttf)
  add_library(openatc_ui_plugin src/ui.cpp src/ui_pages.cpp src/ui_maps.cpp src/ui_settings.cpp src/speech.cpp "${OPENATC_GENERATED_DIR}/openatc_fonts.hpp")
  target_include_directories(openatc_ui_plugin PRIVATE ${miniaudio_SOURCE_DIR} "${OPENATC_GENERATED_DIR}")
  target_link_libraries(openatc_ui_plugin PUBLIC openatc_transport_plain openatc_imgui ${CMAKE_DL_LIBS})
  add_library(openatc_ui ALIAS openatc_ui_plugin)
  if(APPLE)
    target_link_libraries(openatc_ui_plugin PUBLIC "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox")
  elseif(WIN32)
    target_link_libraries(openatc_ui_plugin PUBLIC winmm ole32)
  endif()
endif()
if(OPENATC_BUILD_DESKTOP)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(GLFW_BUILD_WAYLAND ON CACHE BOOL "" FORCE)
    set(GLFW_BUILD_X11 OFF CACHE BOOL "" FORCE)
  endif()
  FetchContent_Declare(glfw URL https://github.com/glfw/glfw/archive/refs/tags/3.4.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(glfw)
  set(OPENATC_PLATFORM_BACKEND ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp)
  set(OPENATC_RENDERER_BACKEND ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl2.cpp)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(OPENATC_PLATFORM_BACKEND src/wayland_input.cpp)
    file(READ "${OPENATC_RENDERER_BACKEND}" OPENATC_RENDERER_SOURCE)
    string(REPLACE "#include <GL/gl.h>" "#include <glad/gl.h>" OPENATC_RENDERER_SOURCE "${OPENATC_RENDERER_SOURCE}")
    if(OPENATC_RENDERER_SOURCE MATCHES "#include <GL/gl.h>")
      message(FATAL_ERROR "Unexpected renderer header dependency")
    endif()
    set(OPENATC_RENDERER_BACKEND "${CMAKE_CURRENT_BINARY_DIR}/imgui_opengl2_loader.cpp")
    file(WRITE "${OPENATC_RENDERER_BACKEND}" "${OPENATC_RENDERER_SOURCE}")
  endif()
  add_executable(open-atc src/desktop.cpp ${OPENATC_PLATFORM_BACKEND} ${OPENATC_RENDERER_BACKEND})
  target_link_libraries(open-atc PRIVATE openatc_ui glfw ${OPENATC_GRAPHICS_TARGET})
  install(TARGETS open-atc RUNTIME DESTINATION bin)
endif()
if(OPENATC_BUILD_PLUGIN)
  if(NOT EXISTS "${XPLANE_SDK}/CHeaders/XPLM/XPLMPlugin.h")
    message(FATAL_ERROR "Set XPLANE_SDK to the extracted SDK directory")
  endif()
  add_library(open-atc-plugin MODULE src/plugin.cpp)
  target_link_libraries(open-atc-plugin PRIVATE openatc_ui_plugin ${OPENATC_GRAPHICS_TARGET})
  target_include_directories(open-atc-plugin PRIVATE "${XPLANE_SDK}/CHeaders/XPLM" "${XPLANE_SDK}/CHeaders/Widgets")
  target_compile_definitions(open-atc-plugin PRIVATE XPLM200 XPLM210 XPLM300 XPLM301 XPLM302 XPLM303 XPLM400)
  set_target_properties(open-atc-plugin PROPERTIES PREFIX "" SUFFIX ".xpl")
  if(WIN32)
    target_compile_definitions(open-atc-plugin PRIVATE IBM=1 APL=0 LIN=0)
    target_link_libraries(open-atc-plugin PRIVATE "${XPLANE_SDK}/Libraries/Win/XPLM_64.lib")
    set_target_properties(open-atc-plugin PROPERTIES OUTPUT_NAME win)
  elseif(APPLE)
    target_compile_definitions(open-atc-plugin PRIVATE IBM=0 APL=1 LIN=0)
    target_link_options(open-atc-plugin PRIVATE "-undefined" "dynamic_lookup")
    set_target_properties(open-atc-plugin PROPERTIES OUTPUT_NAME mac)
  else()
    target_compile_definitions(open-atc-plugin PRIVATE IBM=0 APL=0 LIN=1)
    set_target_properties(open-atc-plugin PROPERTIES OUTPUT_NAME lin BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
  endif()
  install(TARGETS open-atc-plugin LIBRARY DESTINATION OpenATC/64 RUNTIME DESTINATION OpenATC/64)
endif()
install(FILES README.md LICENSE DESTINATION .)
install(DIRECTORY docs assets DESTINATION .)
